const int Y1 = 8;
const int Y2 = 21;
const int Y3 = 41;

const int X1  = 48;
const int X2  = 45;
const int X3  = 17;
const int X4  = 16;
const int X5  = 40;
const int X6  = 18;
const int X7  = 42;
const int X8  = 39;
const int X9  = 47;
const int X10 = 7;

const int linhas[3]  = {Y1, Y2, Y3};
const int colunas[10] = {X1, X2, X3, X4, X5, X6, X7, X8, X9, X10};

bool matriz[3][10]; // [linha][coluna]

void setup() {
  for (int l = 0; l < 3; l++) {
    pinMode(linhas[l], INPUT_PULLUP);
  }

  // Colunas comecam em alta impedancia (nao afetam nada ate serem ativadas)
  for (int c = 0; c < 10; c++) {
    pinMode(colunas[c], INPUT);
  }

  Serial.begin(9600);
}

void loop() {
  for (int c = 0; c < 10; c++) {
    // Ativa so essa coluna
    pinMode(colunas[c], OUTPUT);
    digitalWrite(colunas[c], LOW);

    delayMicroseconds(5); // pequeno tempo para o sinal estabilizar

    for (int l = 0; l < 3; l++) {
      matriz[l][c] = (digitalRead(linhas[l]) == LOW);
    }

    // Desativa a coluna de novo (alta impedancia)
    pinMode(colunas[c], INPUT);
  }

  // Imprime a matriz
  for (int l = 0; l < 3; l++) {
    for (int c = 0; c < 10; c++) {
      Serial.print(matriz[l][c] ? "X" : "N");
      Serial.print(" ");
    }
    Serial.print("\n");
  }
  Serial.print("\n\n");

  delay(300);
}
